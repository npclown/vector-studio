fn run(ws: &mut StrokeWorkspace, req: StrokeRequest, verbs: &[u8], bytes: &[u8]) -> String {
    let t = Instant::now();
    let out = ws.build(req, verbs, bytes).expect("selfcheck");
    let us = t.elapsed().as_micros();
    let st = ws.rounded_stats();
    let pieces = ws.pieces().iter().filter(|p| !p.dropped).count();
    let records = ws.pieces().len();
    let emitted: usize = ws.pieces().iter().filter(|p| !p.dropped).map(|p| p.count).sum();
    let status = match out { StrokeOutcome::Mesh => "ok".to_string(), StrokeOutcome::Status(s) => format!("s{s}") };
    let err = format!("{:?}", ws.last_error());
    let tris = match out { StrokeOutcome::Mesh => ws.mesh().map_or(0, |(_, t, _)| t.len() / 3), _ => 0 };
    format!("{status}\t{err}\t{records}\t{pieces}\t{emitted}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\t{tris}\t{us}",
        st.input_vertices, st.edges, st.pair_checks, st.events, st.sections, st.nodes, st.cells, st.work_units)
}
